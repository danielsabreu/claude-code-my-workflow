% Appendix C: Financial cycle confidence intervals — individual country results
%
% Each panel shows the financial cycle indicator with 95\% confidence bands
% for two countries. The shaded area represents the pointwise confidence band.
% Dashed horizontal lines indicate the mean of the upper and lower bounds,
% which are used to classify periods into elevated, neutral, and subdued regimes.

\begin{figure}[htbp]
  \centering
  \includegraphics[width=\linewidth]{panel_ci_1_2.pdf}
  \caption{Financial cycle and confidence intervals: Belgium and Germany}
  \caption*{\footnotesize \textbf{Notes:} The figure displays the estimated financial cycle (solid line) for Belgium (top) and Germany (bottom), together with 95\% confidence bands (shaded area) based on the asymptotic distribution of the PCA estimator. The dashed lines indicate the mean of the upper and lower confidence bounds, which are used to classify financial cycle regimes. Values above (below) the upper (lower) dashed line indicate statistically significant deviations associated with elevated (subdued) cyclical systemic risk.}
  \label{fig:ci_BE_DE}
\end{figure}
