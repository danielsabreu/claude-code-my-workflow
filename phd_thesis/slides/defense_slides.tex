% ----------------------------------------------------------
% SLIDE 3: STFM Motivation
% ----------------------------------------------------------
\begin{frame}{Essay I: Motivation}

\begin{block}{Standard assumption}
  $\boldsymbol{x}_t = \boldsymbol{\Lambda}\boldsymbol{f}_t + \boldsymbol{e}_t$ \quad --- \quad factor loadings $\boldsymbol{\Lambda}$ \textbf{fixed} over time
\end{block}

\vspace{0.7em}

\textbf{Problem:} economic relationships shift across business cycle phases

\vspace{0.3em}
\begin{itemize}
  \item Structural break models: one-off, non-recurrent shifts
  \item Threshold factor models: recurrent, but \textbf{abrupt} transitions
  \item[\textcolor{isegblue}{$\Rightarrow$}] \textcolor{isegblue}{\textbf{This paper:}} loadings evolve \textbf{smoothly} between two regimes
\end{itemize}

\vspace{0.7em}

\begin{block}{This paper: smooth transition factor model}
  Regime-specific loadings $\boldsymbol{\Lambda}_1$, $\boldsymbol{\Lambda}_2$ weighted by a logistic function
  $g(s_t,\gamma,c)\in(0,1)$ — \textbf{gradual}, \textbf{parametric}, \textbf{recurrent}
\end{block}

\end{frame}

% ----------------------------------------------------------
% SLIDE 4: STFM Model
% ----------------------------------------------------------
\begin{frame}{Essay I: Model}

\textbf{Smooth Transition Factor Model (STFM):}
\[
\boldsymbol{x}_t = \bigl(\boldsymbol{I}_{N}-\boldsymbol{G}_{t}\bigr)\boldsymbol{\Lambda}_1\boldsymbol{f}_t
  + \boldsymbol{G}_{t}\boldsymbol{\Lambda}_2\boldsymbol{f}_t + \boldsymbol{e}_t, \quad t=1,\ldots,T
\]

\begin{itemize}
  \item $\boldsymbol{f}_t$: $r \times 1$ vector of latent factors; \;$\boldsymbol{\Lambda}_1,\boldsymbol{\Lambda}_2$: $N\times r$ regime-specific loadings
  \item $\boldsymbol{G}_t = g(s_t,\gamma,c)\,\boldsymbol{I}_N$: diagonal transition matrix
  \[
  g(s_t,\gamma,c) = \frac{1}{1+\exp\{-\gamma(s_t-c)\}}, \quad \gamma>0
  \]
  \item $s_t$: observable transition variable; $c$: threshold; $\gamma$: smoothness
\end{itemize}

\vspace{0.4em}
\textbf{Interpretation:}
\begin{itemize}
  \item $g\approx 0$ (when $s_t \ll c$): loadings $\approx\boldsymbol{\Lambda}_1$ \quad (regime 1, e.g.\ expansion)
  \item $g\approx 1$ (when $s_t \gg c$): loadings $\approx\boldsymbol{\Lambda}_2$ \quad (regime 2, e.g.\ recession)
  \item Nests linear DFM ($\gamma=0$) and threshold factor model ($\gamma\to\infty$)
\end{itemize}

\end{frame}

% ----------------------------------------------------------
% SLIDE 5: STFM Estimation and Theoretical Results
% ----------------------------------------------------------
\begin{frame}{Essay I: Estimation and Theoretical Results}

\textbf{NLS estimator:} \quad $\boldsymbol{\theta}=(\gamma,c)$
\[
(\hat{\boldsymbol{\Lambda}},\hat{\boldsymbol{f}}_t,\hat{\boldsymbol{\theta}}) = \arg\min_{\boldsymbol{\Lambda},\boldsymbol{f}_t,\boldsymbol{\theta}}\;
  \frac{1}{NT}\sum_{t=1}^{T}
  \bigl[\boldsymbol{x}_t - (\boldsymbol{I}_N{-}\boldsymbol{G}_t)\boldsymbol{\Lambda}_1\boldsymbol{f}_t - \boldsymbol{G}_t\boldsymbol{\Lambda}_2\boldsymbol{f}_t\bigr]'[\cdots]
\]

\textbf{Procedure:} profile over a grid $\boldsymbol{\Gamma}=[0.1,6]\times [\underline{c},\bar{c}]$ (trimming $\tau=0.15$); extract factors via PCA at each grid point; refine with BFGS (5 restarts).

\vspace{0.5em}
\textbf{Theoretical results:}

\begin{tabular}{p{0.45\textwidth} p{0.48\textwidth}}
\textbf{Transition parameters} & \textbf{Factor space} \\[2pt]
$\hat{\gamma},\hat{c}$ consistent as $N,T\to\infty$ &
  PCA estimator consistent at rate $\min\{\sqrt{N},\sqrt{T}\}$ \\
\end{tabular}

\vspace{0.4em}
{\small Using consistency of $\hat{\boldsymbol{\theta}}$, we show the concentrated factor estimator $\hat{\boldsymbol{f}}_t(\hat{\boldsymbol{\Lambda}},\hat{\boldsymbol{\theta}}) = N^{-1}[(\boldsymbol{I}_N-\hat{\boldsymbol{G}}_t)\hat{\boldsymbol{\Lambda}}_1+\hat{\boldsymbol{G}}_t\hat{\boldsymbol{\Lambda}}_2]'\boldsymbol{x}_t$ recovers the true factor space at the standard rate.}

\end{frame}

% ----------------------------------------------------------
% SLIDE 6: Factor Number Selection
% ----------------------------------------------------------
\begin{frame}{Essay I: Selection of the Number of Factors}

\textbf{Key insight (from the structural break literature):}

\begin{block}{}
A single structural break in loadings \textbf{doubles} the required factor dimension: one set of factors for the pre-break subsample, another for the post-break subsample. The same applies to smooth transitions.
\end{block}

\vspace{0.5em}
\textbf{Consequence:} applying the standard Bai--Ng (2002) $IC_1$ to a panel with smooth transition dynamics \textbf{over-estimates} the number of factors.

\vspace{0.4em}
\textbf{Solution:} we develop a \textbf{robustified criterion} $IC_3$, adapted to the smooth transition context:
\begin{itemize}
  \item Accounts for the effective doubling of the factor dimension under regime shifts
  \item Monte Carlo: $IC_3$ selects $\bar{r}\approx 2$ (true $r^0=2$) across all designs
\end{itemize}

\vspace{0.4em}
\textbf{Empirical (FRED-QD):} $IC_3 \Rightarrow \hat{r}=3$ \quad vs.\quad $IC_1 \Rightarrow \tilde{r}=6 \approx 2\hat{r}$, confirming the inflation effect.

\end{frame}

% ----------------------------------------------------------
% SLIDE 7: Linearity Test
% ----------------------------------------------------------
\begin{frame}{Essay I: Linearity Test}

\textbf{Null hypothesis:} $H_0: \gamma=0$ (linear DFM)

\vspace{0.3em}
\textbf{Problem (Davies, 1977):} under $H_0$, the threshold $c$ is \textbf{unidentified} --- standard LM statistics are non-standard.

\vspace{0.4em}
\textbf{Solution:} Taylor-expand $g(s_t;\gamma,c)$ around $\gamma=0$:
\[
g(s_t;\gamma,c) \approx \tfrac{1}{2} + \tfrac{\gamma}{4}(s_t-c) + O(\gamma^3)
\]
Under $H_0$ the nuisance parameter $c$ \textbf{drops out}. The model reduces to testing $\boldsymbol{\delta}=\boldsymbol{0}$ in:
\[
\boldsymbol{x}_t \approx \boldsymbol{\Lambda}^*\boldsymbol{f}_t + \boldsymbol{\Delta}(\boldsymbol{f}_t \cdot s_t) + \boldsymbol{e}_t
\]

The \textbf{LM statistic} is asymptotically $\chi^2$ and can be bootstrapped to account for cross-sectional dependence.

\vspace{0.5em}
\textbf{Empirical result (FRED-QD):}\\[3pt]
\begin{tabular}{ll}
$LM = 108$ & bootstrapped $p$-value $= 0.006$ (1{,}000 replications)
\end{tabular}\\[3pt]
$\Rightarrow$ \textbf{Strongly reject} linearity at all conventional significance levels.

\end{frame}

% SLIDE 10: NSTFM Motivation

\begin{frame}{Essay II: Motivation}

This paper contributes to the strand of literature that considers the case in which \textbf{factors follow a VECM that admits nonlinear adjustment dynamics}.

\begin{itemize}
  \item Empirical evidence suggests that VECM parameters \textbf{vary between regimes} (Baumeister \& Benati 2013; Liu \& Chen 2020)
  \item \textbf{Threshold cointegration} (Balke \& Fomby 1997): adjustments toward long-run equilibrium are activated only when deviations exceed a threshold, capturing frictions, adjustment costs, or market imperfections
\end{itemize}

\vspace{0.3em}
\textbf{Two specifications:}
\begin{enumerate}
  \item \textbf{Threshold factor VECM} --- asymmetric dynamics across regimes
  \item \textbf{Band factor VECM} --- adjustment \emph{inactive} within band, \emph{active} outside
\end{enumerate}

\vspace{0.2em}
Estimated within a \textbf{two-level factor model} (global + group-specific I(1) factors), extending Breitung \& Eickmeier (2016).

\end{frame}

% ----------------------------------------------------------
% SLIDE 11: Two-Level Model
% ----------------------------------------------------------
\begin{frame}{Essay II: Two-Level Factor Model}

\textbf{Observation equation for variable $i$ in group $r$:}
\[
x_{r,it} = \boldsymbol{\gamma}_{r,i}'\boldsymbol{G}_t + \boldsymbol{\lambda}_{r,i}'\boldsymbol{F}_{r,t} + \xi_{r,it}
\]

In matrix form:\quad $\boldsymbol{x}_{r,t} = \boldsymbol{\Gamma}_r\boldsymbol{G}_t + \boldsymbol{\Lambda}_r\boldsymbol{F}_{r,t} + \boldsymbol{\xi}_t$

\begin{itemize}
  \item $\boldsymbol{G}_t$: $m_0\times1$ vector of I(1) \textbf{global factors} (random walk: $\boldsymbol{G}_t = \boldsymbol{G}_{t-1}+\boldsymbol{v}_t$)
  \item $\boldsymbol{F}_{r,t}$: $m_r\times1$ vector of I(1) \textbf{group-specific factors} (random walk)
\end{itemize}

\vspace{0.5em}
\textbf{Key question:} Are the stacked group-specific factors
$\boldsymbol{\mathcal{F}}_t = (\boldsymbol{F}_{1,t}',\ldots,\boldsymbol{F}_{R,t}')'$
cointegrated, and if so, is adjustment \textbf{nonlinear}?

\end{frame}

% ----------------------------------------------------------
% SLIDE 12: Threshold and Band VECM
% ----------------------------------------------------------
\begin{frame}{Essay II: Threshold and Band Factor VECM}

Let $z_{t-1} = \boldsymbol{\beta}'\boldsymbol{\mathcal{F}}_{t-1}$ (cointegration error).

\vspace{0.4em}
\textbf{Threshold factor VECM:}
\[
\Delta\boldsymbol{\mathcal{F}}_t = \begin{cases}
\boldsymbol{\alpha}_1\boldsymbol{\beta}'\boldsymbol{\mathcal{F}}_{t-1} + \boldsymbol{\Phi}_1(L)\Delta\boldsymbol{\mathcal{F}}_{t-1} + \boldsymbol{u}_{1t}, & z_{t-1}\le\gamma\\[3pt]
\boldsymbol{\alpha}_2\boldsymbol{\beta}'\boldsymbol{\mathcal{F}}_{t-1} + \boldsymbol{\Phi}_2(L)\Delta\boldsymbol{\mathcal{F}}_{t-1} + \boldsymbol{u}_{2t}, & z_{t-1}>\gamma
\end{cases}
\]

\textbf{Band factor VECM} (inactive inside band):
\[
\Delta\boldsymbol{\mathcal{F}}_t = \begin{cases}
\boldsymbol{\Phi}_2(L)\Delta\boldsymbol{\mathcal{F}}_{t-1} + \boldsymbol{u}_t, & |z_{t-1}|\le\gamma\\[3pt]
\boldsymbol{\alpha}\boldsymbol{\beta}'\boldsymbol{\mathcal{F}}_{t-1} + \boldsymbol{\Phi}_1(L)\Delta\boldsymbol{\mathcal{F}}_{t-1} + \boldsymbol{u}_t, & |z_{t-1}|>\gamma
\end{cases}
\]

\textbf{Estimation:} SVD initialisation $\to$ sequential least squares for factors $\to$ joint grid search over $(\boldsymbol{\beta},\gamma)$ using reduced-rank regression at each grid point, selecting $(\hat{\boldsymbol{\beta}},\hat{\gamma})$ minimising SSR.

\end{frame}

\begin{frame}{Essay III: Motivation}

The financial cycle captures \textit{``self-reinforcing interactions between perceptions of value, risk-taking behavior, and financing constraints, which typically manifest as periods of credit booms followed by busts''} (Borio, 2014).

\vspace{0.5em}
Accurate measurement is key for macroprudential policy, in particular the calibration of the \textbf{Countercyclical Capital Buffer (CCyB)}: capital requirements that should \emph{increase} during expansionary phases and be \emph{released} during downturns.

\vspace{0.5em}
\textbf{Limitations of the Basel III benchmark (credit-to-GDP gap):}
\begin{itemize}
  \item Relies on the HP filter ($\lambda=400{,}000$) --- large end-of-sample bias; upward phase overly incorporated into trend
  \item Uses only one variable: misses asset-price and leverage dynamics
\end{itemize}

\vspace{0.4em}
\textbf{This paper:} A DFM that extracts a single latent factor summarising the co-movement of credit, asset prices, and other financial indicators for 10 European countries.

\end{frame}

% SLIDE 20: FC Methodology
% ----------------------------------------------------------
\begin{frame}{Essay III: DFM Framework}

\textbf{Static factor model (estimated country by country):}
\[
\boldsymbol{x}_t = \boldsymbol{\Lambda}\boldsymbol{f}_t + \boldsymbol{e}_t
\]

$\boldsymbol{x}_t=[x_{1t},\ldots,x_{Nt}]'$: $N=7$ standardised financial indicators;\quad $\boldsymbol{f}_t$: $r\times1$ latent factors;\\
$\boldsymbol{\Lambda}$: $N\times r$ factor loadings;\quad $\boldsymbol{e}_t$: idiosyncratic, uncorrelated with $\boldsymbol{f}_t$.

\vspace{0.3em}
\textbf{Factor dynamics:} $\boldsymbol{f}_t = \boldsymbol{\Phi}_1\boldsymbol{f}_{t-1}+\cdots+\boldsymbol{\Phi}_p\boldsymbol{f}_{t-p}+\boldsymbol{\eta}_t$ \hfill ($p=4$ lags by IC)

\vspace{0.4em}
\textbf{Three estimation methods:}
\begin{enumerate}
  \item \textbf{PCA}: minimises $\frac{1}{NT}\sum_t (\boldsymbol{x}_t-\boldsymbol{\Lambda}\boldsymbol{f}_t)'(\boldsymbol{x}_t-\boldsymbol{\Lambda}\boldsymbol{f}_t)$
  \item \textbf{QML}: Kalman filter + EM algorithm (state-space representation)
  \item \textbf{Two-step (TSTEP)}: PCA initialisation $\to$ Kalman smoother
\end{enumerate}

Single factor ($r=1$) used in baseline; justified by likelihood-ratio test.

\end{frame}

\begin{frame}{Contributions of the Thesis}

\begin{block}{Essay I --- Smooth Transition Factor Model}
  Consistent NLS+PCA estimator for $\boldsymbol{x}_t=(\boldsymbol{I}_N-\boldsymbol{G}_t)\boldsymbol{\Lambda}_1\boldsymbol{f}_t+\boldsymbol{G}_t\boldsymbol{\Lambda}_2\boldsymbol{f}_t+\boldsymbol{e}_t$; linearity test; $IC_3$ factor selection. Nonlinear factors improve FRED-QD forecasts ($N=231$, $T=133$).
\end{block}

\vspace{0.3em}

\begin{block}{Essay II --- Nonstationary DFM with Threshold Cointegration}
  Threshold and band factor VECM in two-level I(1) models; joint grid search over $(\boldsymbol{\beta},\gamma)$. Global yield-curve factors show discontinuous adjustment ($\hat{\gamma}=0.46$, $\approx55\%$ inside band).
\end{block}

\vspace{0.3em}

\begin{block}{Essay III --- Financial Cycle Measurement}
  DFM indicator (7 variables, 10 EU countries, 1995--2024); AUROC 0.66 vs.\ Basel gap 0.56; 80\% confidence bands define neutral CCyB environment; out-of-sample crisis probability signals GFC well in advance.
\end{block}

\end{frame}

\begin{frame}{Common Thread and Future Research}

\textbf{All three essays relax a restrictive assumption in the standard linear DFM:}

\vspace{0.3em}
{\small
\begin{tabular}{lll}
\toprule
\textbf{Essay} & \textbf{Assumption relaxed} & \textbf{How} \\
\midrule
I   & Constant loadings     & Logistic transition matrix $\boldsymbol{G}_t$ \\
II  & Linear adjustment     & Threshold / band VECM on $\boldsymbol{\mathcal{F}}_t$ \\
III & Single indicator      & Multi-variable latent factor \\
\bottomrule
\end{tabular}
}

\vspace{0.5em}
\textbf{Future research:}
\begin{itemize}
  \item \textbf{Asymptotic distribution theory} for the NLS estimators in Essays I and II (confidence intervals for $(\hat\gamma,\hat c)$ and $(\hat{\boldsymbol{\beta}},\hat\gamma)$)
  \item \textbf{Impulse response functions} under nonlinear factor VECMs
  \item \textbf{Multiple regimes} ($K>2$) with endogenous regime selection
  \item \textbf{Real-time financial cycle monitoring} with mixed-frequency data
\end{itemize}
