\begin{frame}{Introduction}

\structure{Motivation}
\begin{itemize}
    \item The Global Financial Crisis placed the concept of the \structure{financial cycle} at the forefront of academic and policy discussions.
    \item Accurately measuring and understanding the financial cycle is essential for effective \structure{macroprudential policy}.
\end{itemize}

\vspace{0.2cm}
\structure{Approach and Contribution}
\begin{itemize}
    \item We construct country-level financial cycle indicators for European economies using a Dynamic Factor Model (DFM).
    \item The proposed indicator:
    \begin{itemize}
        \item tracks cyclical systemic risk and identifies key drivers over time $\Rightarrow$ \structure{risk monitoring};
        \item provides thresholds to define neutral risk environments and delivers early-warning signals $\Rightarrow$ \structure{supports policy}.
    \end{itemize}
\end{itemize}

\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Introduction}

\structure{Definition (Bank of England, 2023):} 
\begin{itemize}
    \item Financial cycles typically involve an expansion of credit and asset prices, greater than that justified by underlying fundamentals. 
\end{itemize}

\structure{Core stylized facts (Borio, 2012):}

\begin{enumerate}
    \item It is most parsimoniously described in terms of credit and property prices.
    \item It has a much lower frequency than the traditional business cycle.
    \item Its peaks are closely associated with financial stress.
    \item It helps detect financial distress with a good lead in real time.
    \item Its length and amplitude depend on policies that affect the financial sector (monetary policy, macroprudential policy).
    
\end{enumerate}
\end{frame}

\begin{frame}{Representation}

\structure{Dynamic Factor Model (DFM):}

Let $\boldsymbol{x}_t$ denote a stationary $N$-dimensional vector process. The model is specified as:
\begin{align*}
    \boldsymbol{x}_t &= \boldsymbol{\Lambda} \boldsymbol{f}_t + \boldsymbol{e}_t \\
    \boldsymbol{f}_t &= \boldsymbol{\Psi}_1 \boldsymbol{f}_{t-1} + \dots + \boldsymbol{\Psi}_p \boldsymbol{f}_{t-p} + \boldsymbol{\eta}_t
\end{align*}

\begin{itemize}
    \item $\boldsymbol{f}_t$: $q \times 1$ vector of latent common factors.
    \item $\boldsymbol{e}_t$: idiosyncratic component, possibly serially and cross-sectionally correlated.
    \item $\boldsymbol{\Lambda}$: $N \times q$ matrix of factor loadings.
    \item $\boldsymbol{\Psi}_j$: $q \times q$ autoregressive coefficient matrices.
    \item $\boldsymbol{\eta}_t$: serially uncorrelated, mean-zero factor innovations.
\end{itemize}

\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{Principal Components Estimation}

\structure{Estimation strategy:} Solve for $\boldsymbol{\Lambda}$ and $\boldsymbol{F}_t$ by minimizing the average squared residuals (divided by NT).
\begin{equation}
    \min_{\boldsymbol{\Lambda}, \boldsymbol{F}_t} \; V_r(\boldsymbol{\Lambda}, \boldsymbol{F}) 
    = \frac{1}{NT} \sum_{t=1}^{T} (\boldsymbol{x}_t - \boldsymbol{\Lambda} \boldsymbol{F}_t)' (\boldsymbol{x}_t - \boldsymbol{\Lambda} \boldsymbol{F}_t)
\end{equation}

\structure{Estimators:} Assuming $N^{-1} \boldsymbol{\Lambda}' \boldsymbol{\Lambda} = \boldsymbol{I}_r$ and $ \Sigma_F = \mathbb{E}[\boldsymbol{F}_t \boldsymbol{F}_t']$ is diagonal, it can be shown that: 

\begin{equation}
\begin{aligned}
    \widehat{\boldsymbol{\Lambda}} &= \text{first $q$ eigenvectors of } \widehat{\boldsymbol{\Sigma}}_x = T^{-1} \sum_{t=1}^{T} \boldsymbol{x}_t \boldsymbol{x}_t' \\
        \widehat{\boldsymbol{F}}_t &= N^{-1} \widehat{\boldsymbol{\Lambda}}' \boldsymbol{x}_t 
\end{aligned}
\end{equation}

\structure{Factor dynamics:} Parameters of VAR model for the factors can be estimated by OLS conditionally on $\widehat{\boldsymbol{F}}_t$.  

\end{frame}

\begin{frame}{Conclusions}

\structure{Final remarks}
    \vspace{0.4cm}
\begin{itemize}
    \item We use factor models to propose a country specific financial cycle indicator for countries across Europe. 
    \vspace{0.4cm}
    \item The measure is able to explain a large share of the variance in credit and asset prices and can be used to assess the evolution of cyclical systemic risk.
\vspace{0.4cm}
    \item The measure provides a reasonable description of the evolution of the financial cycle and demonstrates good early-warning properties concerning the Great financial crisis.
\end{itemize}
